import numpy as np
import pandas as pd

#Pandas DataFrame Creation
data={
	"Name":["Manish", "Amit", "Riya", "Sourav", "Neha", "Rahul"],
	"Roll_Number":[10,11, 12, 13, 14, 15],
	"Attendance":[88, 92, 76, 95, 81, 67],
	"Marks":[72, 85, 68, 91, 77, 95]
}

df = pd.DataFrame(data)

# Function to calculate Grade based on Marks
def calculate_grade(marks):
    if marks >= 90:
        return "A+"
    elif marks >= 80:
        return "A"
    elif marks >= 70:
        return "B"
    elif marks >= 60:
        return "C"
    elif marks >= 50:
        return "D"
    else:
        return "F"

df["Grade"] = df["Marks"].apply(calculate_grade)
print(df)
